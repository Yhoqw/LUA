=======
LEGEND
=======

. = Background
# = Wall
O = Rock
X = Crate
P = PlayerDown
_ = Pad
D = Gate
E = Exit
G = Guard

Player = PlayerUp or PlayerDown or PlayerLeft or PlayerRight
Solid = Wall or Rock or Crate or Bomb1 or Bomb2 or Bomb3 or Guard or Gate
BlastBlock = Wall or Rock or Crate or Gate
HotFire = Blast or BlastUp or BlastUp2 or BlastDown or BlastDown2 or BlastLeft or BlastLeft2 or BlastRight or BlastRight2
Fire = HotFire or FireOld

======
RULES
======

left [ > Player ] -> [ > PlayerLeft ]
right [ > Player ] -> [ > PlayerRight ]
up [ > Player ] -> [ > PlayerUp ]
down [ > Player ] -> [ > PlayerDown ]

[ > Player | Crate ] -> [ > Player | > Crate ]

left [ action PlayerLeft | no Solid ] -> [ PlayerLeft | Bomb3 Fresh ] sfx0
right [ action PlayerRight | no Solid ] -> [ PlayerRight | Bomb3 Fresh ] sfx0
up [ action PlayerUp | no Solid ] -> [ PlayerUp | Bomb3 Fresh ] sfx0
down [ action PlayerDown | no Solid ] -> [ PlayerDown | Bomb3 Fresh ] sfx0

late [ HotFire ] -> [ FireOld ]

late [ Bomb1 no Fresh ] -> [ Blast ] sfx1
late [ Bomb2 no Fresh ] -> [ Bomb1 ]
late [ Bomb3 no Fresh ] -> [ Bomb2 ]
late [ Fresh ] -> []

late [ Fire Bomb1 ] -> [ Blast ]
+ late [ Fire Bomb2 ] -> [ Blast ]
+ late [ Fire Bomb3 ] -> [ Blast ]
+ late [ Fire | Bomb1 ] -> [ Fire | Blast ]
+ late [ Fire | Bomb2 ] -> [ Fire | Blast ]
+ late [ Fire | Bomb3 ] -> [ Fire | Blast ]
+ late up [ Blast | no BlastBlock no HotFire ] -> [ Blast | BlastUp ]
+ late up [ BlastUp | no BlastBlock no HotFire ] -> [ BlastUp | BlastUp2 ]
+ late down [ Blast | no BlastBlock no HotFire ] -> [ Blast | BlastDown ]
+ late down [ BlastDown | no BlastBlock no HotFire ] -> [ BlastDown | BlastDown2 ]
+ late left [ Blast | no BlastBlock no HotFire ] -> [ Blast | BlastLeft ]
+ late left [ BlastLeft | no BlastBlock no HotFire ] -> [ BlastLeft | BlastLeft2 ]
+ late right [ Blast | no BlastBlock no HotFire ] -> [ Blast | BlastRight ]
+ late right [ BlastRight | no BlastBlock no HotFire ] -> [ BlastRight | BlastRight2 ]

late [ Blast | Rock ] -> [ Blast | ]
late up [ BlastUp | Rock ] -> [ BlastUp | ]
late down [ BlastDown | Rock ] -> [ BlastDown | ]
late left [ BlastLeft | Rock ] -> [ BlastLeft | ]
late right [ BlastRight | Rock ] -> [ BlastRight | ]

late [ Fire Guard ] -> [ Fire ]
late [ Fire Player ] -> restart sfx1

late [ Pad Crate | Gate ] -> [ Pad Crate | ] sfx2

late [ FireOld ] -> []
